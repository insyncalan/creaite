   ```python
   #!/usr/bin/env python3
   import glob, json, os, subprocess, sys
   from pathlib import Path
   args = sys.argv[1:]
   msg = args[args.index("-p") + 1] if "-p" in args else ""
   vault, eng = Path.cwd(), os.environ.get("HC_ENG", "")
   hook = json.loads(args[args.index("--settings") + 1])["hooks"]["PreToolUse"][0]["hooks"][0]["command"] if "--settings" in args else None
   out = lambda o: print(json.dumps(o), flush=True)
   n = [0]
   def call(tool, inp, real):
       n[0] += 1
       tid = f"t{n[0]}"
       out({"type": "assistant", "message": {"content": [{"type": "tool_use", "id": tid, "name": tool, "input": inp}]}})
       r = subprocess.run(hook, shell=True, input=json.dumps({"hook_event_name": "PreToolUse", "tool_name": tool, "tool_input": inp}),
                          capture_output=True, text=True, cwd=vault) if hook else None
       res = r.stderr if r is not None and r.returncode == 2 else real()
       out({"type": "user", "message": {"content": [{"type": "tool_result", "tool_use_id": tid, "content": res}]}})
       return res
   out({"type": "system", "subtype": "init", "session_id": "sess-hc"})
   notes = sorted(glob.glob(f"{eng}/*/08 Notes/*.md"))
   heard = sorted(glob.glob(f"{eng}/*/02 Meetings/Transcripts/*"))
   src = (heard if "transcript" in msg else notes)[0]
   call("Read", {"file_path": src}, lambda: Path(src).read_text(encoding="utf-8"))
   draft = vault / "system/staging/drafts/hc-follow-up.md"
   call("Write", {"file_path": str(draft), "content": "Follow up with Omar.\n"},
        lambda: (draft.write_text("Follow up with Omar.\n"), "written")[1])
   target = str(Path(notes[0]).parent / "HC follow-up.md")
   cmd = f'python3 system/scripts/filechange.py create "{target}" --from system/staging/drafts/hc-follow-up.md --why "Alan asked"'
   call("Bash", {"command": cmd}, lambda: (lambda r: r.stdout + r.stderr)(subprocess.run(cmd, shell=True, capture_output=True, text=True, cwd=vault)))
   text = f"Read {Path(src).name}; asked to make HC follow-up.md."
   out({"type": "assistant", "message": {"content": [{"type": "text", "text": text}]}})
   out({"type": "result", "session_id": "sess-hc", "result": text})
   ```
